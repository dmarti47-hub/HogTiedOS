cmake_minimum_required(VERSION 3.16)
project(libhbas C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_EXTENSIONS ON)          # tests use statement expressions

add_library(hbas STATIC src/vehicle.c src/ioc.c src/buttons.c src/audio.c src/eq.c src/settings.c
  src/dsp.c src/harley_eq_table.c src/bike.c)
target_include_directories(hbas PUBLIC include)
target_link_libraries(hbas PUBLIC m)
target_compile_options(hbas PRIVATE -Wall -Wextra -Werror)

option(HBAS_TESTS "Build libhbas unit tests" ON)
if(HBAS_TESTS)
  enable_testing()
  add_executable(test_vehicle tests/test_vehicle.c)
  target_link_libraries(test_vehicle hbas)
  target_compile_options(test_vehicle PRIVATE -Wall -Wextra -Werror)
  add_test(NAME vehicle COMMAND test_vehicle)
  add_executable(test_ioc tests/test_ioc.c)
  target_link_libraries(test_ioc hbas)
  target_compile_options(test_ioc PRIVATE -Wall -Wextra -Werror)
  add_test(NAME ioc COMMAND test_ioc)
  add_executable(test_buttons tests/test_buttons.c)
  target_link_libraries(test_buttons hbas)
  target_compile_options(test_buttons PRIVATE -Wall -Wextra -Werror)
  add_test(NAME buttons COMMAND test_buttons)
  add_executable(test_audio tests/test_audio.c)
  target_link_libraries(test_audio hbas)
  target_compile_options(test_audio PRIVATE -Wall -Wextra -Werror)
  add_test(NAME audio COMMAND test_audio)
  add_executable(test_eq tests/test_eq.c)
  target_link_libraries(test_eq hbas)
  target_compile_options(test_eq PRIVATE -Wall -Wextra -Werror)
  add_test(NAME eq COMMAND test_eq)
  add_executable(test_settings tests/test_settings.c)
  target_link_libraries(test_settings hbas)
  target_compile_options(test_settings PRIVATE -Wall -Wextra -Werror)
  add_test(NAME settings COMMAND test_settings)
  add_executable(test_dsp tests/test_dsp.c)
  target_link_libraries(test_dsp hbas)
  target_compile_options(test_dsp PRIVATE -Wall -Wextra -Werror)
  add_test(NAME dsp COMMAND test_dsp)
endif()

install(TARGETS hbas ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
